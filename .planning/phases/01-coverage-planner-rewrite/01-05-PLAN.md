9. `fetch_all_areas`: poll-without-spin pattern from PATTERNS.md §"Service Poll Without Spin" (use the helper_node from coverage_nodes.cpp:64-86 only if applicable; here we can use a 5-second timeout in a sleep-poll loop because the node's executor IS the one we're running on, and the action server's `execute()` is in a detached thread — so the executor IS still spinning callback delivery while we wait):
```cpp
if (!get_all_areas_client_->wait_for_service(std::chrono::seconds(5))) {
  ctx.error = mowgli_interfaces::msg::PlanError{};
  ctx.error->error_code = mowgli_interfaces::msg::PlanError::ERROR_INTERNAL;
  ctx.error->human_readable = "GetAllAreas service not available";
  return false;
}
auto req = std::make_shared<mowgli_interfaces::srv::GetAllAreas::Request>();
auto fut = get_all_areas_client_->async_send_request(req);
auto deadline = std::chrono::steady_clock::now() + std::chrono::seconds(5);
while (rclcpp::ok() && !goal_handle->is_canceling()) {
  if (fut.wait_for(std::chrono::milliseconds(20)) == std::future_status::ready) {
    ctx.areas = fut.get()->areas;
    return true;
  }
  if (std::chrono::steady_clock::now() > deadline) break;
}
ctx.error = mowgli_interfaces::msg::PlanError{};
ctx.error->error_code = mowgli_interfaces::msg::PlanError::ERROR_INTERNAL;
ctx.error->human_readable = "GetAllAreas timed out";
return false;
```

```cpp
void CoveragePlannerNode::on_write_checkpoint(
    const mowgli_interfaces::srv::WriteCheckpoint::Request::SharedPtr req,
    mowgli_interfaces::srv::WriteCheckpoint::Response::SharedPtr res)
{
  std::string err;
  if (!write_checkpoint_file(areas_dir_, req->checkpoint, &err)) {
    res->success = false;
    res->error_message = err;
    RCLCPP_ERROR(get_logger(), "WriteCheckpoint failed: %s", err.c_str());
    return;
  }
  res->success = true;
  res->error_message.clear();
}
```
