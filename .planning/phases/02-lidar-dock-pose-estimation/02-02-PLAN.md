- `save_dock_scan_pcd`: build `pcl::PointCloud<pcl::PointXYZ>`, set width/height/is_dense per the snippet, call `pcl::io::savePCDFile(path, cloud, false)` (binary=false → ASCII per D-06). Return true iff retval == 0.

Return true iff atomic_write returns true. This delivers SPEC R-11 power-loss-safe refresh.

```cpp
#pragma once
#include <mowgli_geometry/key_value_parser.hpp>
#include <optional>
#include <string>

namespace mowgli_lidar_docking {

struct DockApproach {
  double x{0.0};
  double y{0.0};
  double yaw_to_dock_rad{0.0};
  std::string source;       // "lidar" | "tf"
  std::string captured_at;  // ISO-8601
};

inline std::optional<DockApproach> load_dock_approach_yaml(const std::string& path) {
  std::ifstream f(path);
  if (!f.good()) return std::nullopt;
  std::stringstream ss; ss << f.rdbuf();
  const std::string content = ss.str();
  auto x = mowgli_geometry::parse_yaml_double(content, "dock_approach_x");
  auto y = mowgli_geometry::parse_yaml_double(content, "dock_approach_y");
  auto yaw = mowgli_geometry::parse_yaml_double(content, "dock_approach_yaw_to_dock_rad");
  auto source = mowgli_geometry::parse_yaml_string(content, "source");
  auto captured = mowgli_geometry::parse_yaml_string(content, "captured_at");
  if (!x || !y || !yaw || !source || !captured) return std::nullopt;
  if (*source != "lidar" && *source != "tf") return std::nullopt;
  return DockApproach{*x, *y, *yaw, *source, *captured};
}

bool save_dock_approach_yaml(const std::string& path, const DockApproach& approach);

}  // namespace mowgli_lidar_docking
```

```cpp
#include "mowgli_lidar_docking/confidence_metrics.hpp"
// PROBE.md A2: include kiss_icp/core/VoxelHashMap.hpp (path resolved from submodule layout in PROBE.md).
#include <kiss_icp/core/VoxelHashMap.hpp>
#include <cmath>

namespace mowgli_lidar_docking {

ConfidenceResult compute_confidence(
    const std::vector<Eigen::Vector3d>& frame,
    const kiss_icp::VoxelHashMap& voxel_map,
    double max_d)
{
  if (frame.empty()) return {0.0, std::numeric_limits<double>::infinity()};
  size_t inliers = 0;
  double sse = 0.0;
  const double max_d2 = max_d * max_d;
  for (const auto& p : frame) {
    // PROBE.md A2: GetClosestNeighbor signature confirmed.
    const Eigen::Vector3d nn = voxel_map.GetClosestNeighbor(p);
    const double d2 = (p - nn).squaredNorm();
    if (d2 <= max_d2) {
      ++inliers;
      sse += d2;
    }
  }
  if (inliers == 0) return {0.0, std::numeric_limits<double>::infinity()};
  return {static_cast<double>(inliers) / static_cast<double>(frame.size()),
          std::sqrt(sse / static_cast<double>(inliers))};
}

ConfidenceResult compute_confidence_brute_force(
    const std::vector<Eigen::Vector3d>& frame,
    const std::vector<Eigen::Vector3d>& reference_cloud,
    double max_d)
{
  if (frame.empty()) return {0.0, std::numeric_limits<double>::infinity()};
  size_t inliers = 0;
  double sse = 0.0;
  const double max_d2 = max_d * max_d;
  for (const auto& p : frame) {
    double best_d2 = std::numeric_limits<double>::infinity();
    for (const auto& q : reference_cloud) {
      const double d2 = (p - q).squaredNorm();
      if (d2 < best_d2) best_d2 = d2;
    }
    if (best_d2 <= max_d2) {
      ++inliers;
      sse += best_d2;
    }
  }
  if (inliers == 0) return {0.0, std::numeric_limits<double>::infinity()};
  return {static_cast<double>(inliers) / static_cast<double>(frame.size()),
          std::sqrt(sse / static_cast<double>(inliers))};
}

bool is_trusted(const ConfidenceResult& r, double min_inlier, double max_rmse) {
  return std::isfinite(r.rmse_m) && r.inlier_ratio >= min_inlier && r.rmse_m <= max_rmse;
}

}  // namespace mowgli_lidar_docking
```
